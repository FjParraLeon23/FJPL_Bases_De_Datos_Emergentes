---Tenemos que usar primero un find. En fin pero la idea base esta bien. Ahora podemos hacer una función para automatizar esto.


function updateByRange(categoria, proveedor_id, skip, limit) {
  db.productos.find({ categoria: categoria })
    .sort({ sku: 1 })
    .skip(skip)
    .limit(limit)
    .forEach(doc => {
      db.productos.updateOne({ _id: doc._id }, { $set: { proveedor_id: proveedor_id } });
    });
}

// Accesorios
updateByRange("Accesorios", "PR001", 0, 5);
updateByRange("Accesorios", "PR002", 5, 5);
updateByRange("Accesorios", "PR003", 10, 4);
